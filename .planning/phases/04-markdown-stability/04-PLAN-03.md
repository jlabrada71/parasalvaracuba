Create `tests/integration/vote-race-condition.spec.ts`:

```typescript
import { describe, it, expect, beforeEach } from 'vitest'
import { query } from '~/server/database'
import fetch from 'node-fetch'

const BASE_URL = 'http://localhost:3000'

describe('Vote Race Condition - Integration Tests', () => {
  let ideaId: string
  let commentId: string
  const testEmail = 'race-test@example.com'
  const voterEmail = 'voter-race-test@example.com'

  beforeEach(async () => {
    // Create test idea
    const ideaResult = await query(
      'INSERT INTO ideas (title, description, email) VALUES ($1, $2, $3) RETURNING id',
      ['Race Test Idea', 'Test description', testEmail]
    )
    ideaId = ideaResult.rows[0].id

    // Create test comment
    const commentResult = await query(
      'INSERT INTO comments (idea_id, email, body) VALUES ($1, $2, $3) RETURNING id',
      [ideaId, testEmail, 'Test comment']
    )
    commentId = commentResult.rows[0].id
  })

  it('should prevent duplicate votes on ideas with concurrent requests', async () => {
    // Send 10 concurrent vote requests from same user
    const votePromises = Array.from({ length: 10 }).map((_, i) => {
      const voteValue = i % 2 === 0 ? 1 : -1  // Mix of +1 and -1
      return fetch(`${BASE_URL}/api/ideas/${ideaId}/votes`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Cookie': `email=${voterEmail}`
        },
        body: JSON.stringify({ vote_value: voteValue })
      })
    })

    // Wait for all requests to complete
    const responses = await Promise.all(votePromises)

    // Verify all responses were successful
    for (const response of responses) {
      expect(response.status).toBe(200)
    }

    // Query database to count votes from this user on this idea
    const result = await query(
      'SELECT COUNT(*) as count FROM votes WHERE idea_id = $1 AND email = $2',
      [ideaId, voterEmail]
    )

    const voteCount = parseInt(result.rows[0].count, 10)

    // Critical assertion: exactly one vote, not 10
    expect(voteCount).toBe(1)
    console.log(`✓ Concurrent 10 votes resulted in exactly ${voteCount} row (no duplicates)`)
  })

  it('should prevent duplicate votes on comments with concurrent requests', async () => {
    // Send 10 concurrent vote requests on comment
    const votePromises = Array.from({ length: 10 }).map((_, i) => {
      const voteValue = i % 2 === 0 ? 1 : -1
      return fetch(`${BASE_URL}/api/comments/${commentId}/votes`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Cookie': `email=${voterEmail}`
        },
        body: JSON.stringify({ vote_value: voteValue })
      })
    })

    const responses = await Promise.all(votePromises)

    for (const response of responses) {
      expect(response.status).toBe(200)
    }

    // Verify only one vote exists
    const result = await query(
      'SELECT COUNT(*) as count FROM votes WHERE comment_id = $1 AND email = $2',
      [commentId, voterEmail]
    )

    const voteCount = parseInt(result.rows[0].count, 10)
    expect(voteCount).toBe(1)
    console.log(`✓ Concurrent 10 comment votes resulted in exactly ${voteCount} row (no duplicates)`)
  })

  it('should allow different users to vote without race condition', async () => {
    // 5 different users vote concurrently
    const votePromises = Array.from({ length: 5 }).map((_, i) => {
      const userEmail = `user-${i}@example.com`
      return fetch(`${BASE_URL}/api/ideas/${ideaId}/votes`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Cookie': `email=${userEmail}`
        },
        body: JSON.stringify({ vote_value: 1 })
      })
    })

    const responses = await Promise.all(votePromises)

    for (const response of responses) {
      expect(response.status).toBe(200)
    }

    // Verify 5 votes exist (one per user)
    const result = await query(
      'SELECT COUNT(*) as count FROM votes WHERE idea_id = $1',
      [ideaId]
    )

    const voteCount = parseInt(result.rows[0].count, 10)
    expect(voteCount).toBe(5)
    console.log(`✓ 5 different users created 5 distinct votes (no race condition)`)
  })

  it('should update vote_value on upsert (not create duplicate)', async () => {
    // First vote: +1
    const response1 = await fetch(`${BASE_URL}/api/ideas/${ideaId}/votes`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Cookie': `email=${voterEmail}`
      },
      body: JSON.stringify({ vote_value: 1 })
    })
    expect(response1.status).toBe(200)
    const vote1 = await response1.json()
    expect(vote1.vote_value).toBe(1)

    // Second vote: -1 (same user)
    const response2 = await fetch(`${BASE_URL}/api/ideas/${ideaId}/votes`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Cookie': `email=${voterEmail}`
      },
      body: JSON.stringify({ vote_value: -1 })
    })
    expect(response2.status).toBe(200)
    const vote2 = await response2.json()
    expect(vote2.vote_value).toBe(-1)

    // Verify only one vote exists in database
    const result = await query(
      'SELECT COUNT(*) as count, vote_value FROM votes WHERE idea_id = $1 AND email = $2 GROUP BY vote_value',
      [ideaId, voterEmail]
    )

    expect(result.rows.length).toBe(1)  // Only one vote_value group
    expect(result.rows[0].count).toBe('1')  // Count is 1
    expect(parseInt(result.rows[0].vote_value)).toBe(-1)  // Latest value
    console.log(`✓ Upsert correctly updated vote from +1 to -1 (no duplicate)`)
  })
})
```

import { describe, it, expect, beforeEach } from 'vitest'
import { query } from '~/server/database'
import fetch from 'node-fetch'

const BASE_URL = 'http://localhost:3000'

describe('Comment Deletion Cascade - Integration Tests', () => {
  let ideaId: string
  let parentCommentId: string
  let replyCommentId: string
  let nestedReplyCommentId: string

  beforeEach(async () => {
    // Create test idea
    const ideaResult = await query(
      'INSERT INTO ideas (title, description, email) VALUES ($1, $2, $3) RETURNING id',
      ['Deletion Test Idea', 'Test', 'idea@example.com']
    )
    ideaId = ideaResult.rows[0].id

    // Create parent comment
    const parentResult = await query(
      'INSERT INTO comments (idea_id, email, body) VALUES ($1, $2, $3) RETURNING id',
      [ideaId, 'author@example.com', 'Parent comment']
    )
    parentCommentId = parentResult.rows[0].id

    // Create reply to parent
    const replyResult = await query(
      'INSERT INTO comments (idea_id, parent_comment_id, email, body) VALUES ($1, $2, $3, $4) RETURNING id',
      [ideaId, parentCommentId, 'reply@example.com', 'Reply to parent']
    )
    replyCommentId = replyResult.rows[0].id

    // Create nested reply
    const nestedResult = await query(
      'INSERT INTO comments (idea_id, parent_comment_id, email, body) VALUES ($1, $2, $3, $4) RETURNING id',
      [ideaId, replyCommentId, 'nested@example.com', 'Nested reply']
    )
    nestedReplyCommentId = nestedResult.rows[0].id
  })

  it('should cascade delete nested replies when parent is deleted', async () => {
    // Verify structure exists
    let result = await query(
      'SELECT COUNT(*) as count FROM comments WHERE idea_id = $1',
      [ideaId]
    )
    expect(parseInt(result.rows[0].count)).toBe(3)  // parent, reply, nested reply

    // Delete parent comment
    const response = await fetch(
      `${BASE_URL}/api/ideas/${ideaId}/comments/${parentCommentId}`,
      {
        method: 'DELETE',
        headers: {
          'Cookie': `email=author@example.com`
        }
      }
    )
    expect(response.status).toBe(200)

    // Verify all comments deleted
    result = await query(
      'SELECT COUNT(*) as count FROM comments WHERE idea_id = $1',
      [ideaId]
    )
    expect(parseInt(result.rows[0].count)).toBe(0)
    console.log(`✓ Deleting parent comment cascaded delete of 2 nested replies`)
  })

  it('should cascade delete votes on deleted comments', async () => {
    // Add votes on parent, reply, and nested reply from multiple users
    await query(
      'INSERT INTO votes (comment_id, email, vote_value) VALUES ($1, $2, $3)',
      [parentCommentId, 'voter1@example.com', 1]
    )
    await query(
      'INSERT INTO votes (comment_id, email, vote_value) VALUES ($1, $2, $3)',
      [parentCommentId, 'voter2@example.com', -1]
    )
    await query(
      'INSERT INTO votes (comment_id, email, vote_value) VALUES ($1, $2, $3)',
      [replyCommentId, 'voter3@example.com', 1]
    )
    await query(
      'INSERT INTO votes (comment_id, email, vote_value) VALUES ($1, $2, $3)',
      [nestedReplyCommentId, 'voter4@example.com', 1]
    )

    // Verify 4 votes exist
    let result = await query(
      'SELECT COUNT(*) as count FROM votes WHERE comment_id IN ($1, $2, $3)',
      [parentCommentId, replyCommentId, nestedReplyCommentId]
    )
    expect(parseInt(result.rows[0].count)).toBe(4)

    // Delete parent comment
    const response = await fetch(
      `${BASE_URL}/api/ideas/${ideaId}/comments/${parentCommentId}`,
      {
        method: 'DELETE',
        headers: {
          'Cookie': `email=author@example.com`
        }
      }
    )
    expect(response.status).toBe(200)

    // Verify all votes deleted
    result = await query(
      'SELECT COUNT(*) as count FROM votes WHERE comment_id IN ($1, $2, $3)',
      [parentCommentId, replyCommentId, nestedReplyCommentId]
    )
    expect(parseInt(result.rows[0].count)).toBe(0)
    console.log(`✓ Deleting comment cascaded delete of 4 votes on all nested comments`)
  })

  it('should only allow comment author to delete', async () => {
    // Try to delete with different email
    const response = await fetch(
      `${BASE_URL}/api/ideas/${ideaId}/comments/${parentCommentId}`,
      {
        method: 'DELETE',
        headers: {
          'Cookie': `email=wrong-user@example.com`
        }
      }
    )
    expect(response.status).toBe(403)  // Forbidden

    // Verify comment still exists
    const result = await query(
      'SELECT COUNT(*) as count FROM comments WHERE id = $1',
      [parentCommentId]
    )
    expect(parseInt(result.rows[0].count)).toBe(1)
    console.log(`✓ Non-author deletion attempt blocked with 403`)
  })

  it('should return 404 for non-existent comment', async () => {
    const response = await fetch(
      `${BASE_URL}/api/ideas/${ideaId}/comments/non-existent-id`,
      {
        method: 'DELETE',
        headers: {
          'Cookie': `email=author@example.com`
        }
      }
    )
    expect(response.status).toBe(404)
    console.log(`✓ Deleting non-existent comment returns 404`)
  })

  it('should not delete votes on other ideas', async () => {
    // Create another idea and comment
    const otherIdeaResult = await query(
      'INSERT INTO ideas (title, description, email) VALUES ($1, $2, $3) RETURNING id',
      ['Other Idea', 'Test', 'other@example.com']
    )
    const otherIdeaId = otherIdeaResult.rows[0].id

    const otherCommentResult = await query(
      'INSERT INTO comments (idea_id, email, body) VALUES ($1, $2, $3) RETURNING id',

    )
    const otherCommentId = otherCommentResult.rows[0].id

    // Add votes on both comments
    await query(
      'INSERT INTO votes (comment_id, email, vote_value) VALUES ($1, $2, $3)',
      [parentCommentId, 'voter@example.com', 1]
    )
    await query(
      'INSERT INTO votes (comment_id, email, vote_value) VALUES ($1, $2, $3)',
      [otherCommentId, 'voter@example.com', 1]
    )

    // Delete from first idea
    const response = await fetch(
      `${BASE_URL}/api/ideas/${ideaId}/comments/${parentCommentId}`,
      {
        method: 'DELETE',
        headers: {
          'Cookie': `email=author@example.com`
        }
      }
    )
